# Forbidden symbols in the portable archives (core/, protocol/, support/;
# PRD §4.2 portability discipline).
#
# Patterns are extended regex (grep -E), one per line, matched against
# `nm -u --format=just-symbols` output. Anchored matches (^name$) preferred.
# The Makefile strips comment and blank lines before invoking grep -E -f.
#
# Adding a symbol is fine. Removing one requires a concrete reason
# (e.g., platform-callback model was extended to allow it).
#
# memcpy/memset are sanctioned exceptions for bounded caller-provided
# storage and byte encoders. memmove stays forbidden because overlap
# requirements should not appear in the portable hot path.

# Heap allocation
^malloc$
^calloc$
^realloc$
^free$
^aligned_alloc$
^posix_memalign$
^strdup$
^strndup$
^getline$
^getdelim$
^asprintf$

# General libc helpers not allowed in the portable hot path
^memmove$
^qsort$
^rand$
^exit$
^abort$
^__assert_fail$

# File-descriptor / I/O syscalls
^open$
^openat$
^read$
^write$
^close$
^lseek$

# Memory mapping
^mmap$
^munmap$
^mprotect$

# Blocking sleep
^sleep$
^usleep$
^nanosleep$
^clock_nanosleep$

# Stdio
^printf$
^fprintf$
^vfprintf$
^puts$
^fopen$
^fclose$
^fread$
^fwrite$
^perror$
^sprintf$
^snprintf$
^vsprintf$
^vsnprintf$

# Pthread (whole family)
^pthread_

# Wall-clock reads — core gets now_ms from the input snapshot, not from libc
^time$
^gettimeofday$
^clock_gettime$

# Soft-float helper calls: any accidental float/double in portable code
# should trip this gate on targets without hardware FP.
^__adddf3$
^__subdf3$
^__muldf3$
^__divdf3$
^__negdf2$
^__floatsidf$
^__floatunsidf$
^__fixdfsi$
^__fixunsdfsi$
^__extendsfdf2$
^__truncdfsf2$
^__addsf3$
^__subsf3$
^__mulsf3$
^__divsf3$
^__negsf2$
^__floatsisf$
^__floatunsisf$
^__fixsfsi$
^__fixunssfsi$
^__aeabi_f(add|sub|mul|div)$
^__aeabi_d(add|sub|mul|div)$
^__aeabi_[ui]*2[fd]$
^__aeabi_[fd]2[ui]*z$
